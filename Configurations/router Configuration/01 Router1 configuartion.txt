enable
configure terminal

! Basic settings
!

!
no ip domain-lookup
!
! Interface configurations
!

!

 no shutdown

 description Link_to_Router-C_Area0

 no shutdown

! OSPF Configuration
!
router ospf 1
 router-id 1.1.1.1
!
 network 10.0.12.0 0.0.0.3 area 0
 network 10.0.13.0 0.0.0.3 area 0
